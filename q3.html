def schedule(jobs):
    total = 0
    maxd = 0

    for i in range(len(jobs)):
        if jobs[i][1] > maxd:
            maxd = jobs[i][1]

    slots = [-1] * maxd

    for i in range(len(jobs)):
        for j in range(i + 1, len(jobs)):
            if jobs[i][2] < jobs[j][2]:
                jobs[i], jobs[j] = jobs[j], jobs[i]

    for i in range(len(jobs)):
        d = jobs[i][1]
        while d > 0:
            if slots[d - 1] == -1:
                slots[d - 1] = jobs[i][0]
                total = total + jobs[i][2]
                break
            d = d - 1

    return total
