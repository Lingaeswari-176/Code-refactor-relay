def longestCommonPrefix(strs):
    if len(strs) == 0:
        return ""

    prefix = strs[0]
    index = 0

    while index < len(prefix):
        ch = prefix[index]

        for word in strs:
            if index >= len(word):
                prefix = prefix[:index]
                return prefix

            if word[index] != ch:
                prefix = prefix[:index]
                return prefix

        index = index + 1

    return prefix
